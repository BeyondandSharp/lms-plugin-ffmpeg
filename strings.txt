	EN	FFmpeg binaries
	DE	FFmpeg-Programme

PLUGIN_FFMPEG_BINARY_SECTION_DESC
	EN	Leave the fields empty to let Lyrion Music Server find ffmpeg in the system PATH. If a path is configured, that exact executable is used.
	DE	Felder leer lassen, damit Lyrion Music Server ffmpeg im System-PATH findet. Ist ein Pfad konfiguriert, wird genau dieses Programm verwendet.

PLUGIN_FFMPEG_FFMPEG_PATH
	EN	ffmpeg executable
	DE	ffmpeg-Programm

PLUGIN_FFMPEG_FFMPEG_PATH_DESC
	EN	Absolute path to the ffmpeg binary, e.g. /usr/bin/ffmpeg.
	DE	Absoluter Pfad zum ffmpeg-Programm, z. B. /usr/bin/ffmpeg.

PLUGIN_FFMPEG_FFPROBE_PATH
	EN	ffprobe executable
	DE	ffprobe-Programm

PLUGIN_FFMPEG_FFPROBE_PATH_DESC
	EN	Absolute path to the ffprobe binary. If empty, ffprobe next to the ffmpeg binary or in the system PATH is used.
	DE	Absoluter Pfad zum ffprobe-Programm. Leer: ffprobe neben ffmpeg oder im System-PATH wird verwendet.

PLUGIN_FFMPEG_FALLBACK_TO_SYSTEM
	EN	Fall back to the system ffmpeg
	DE	Auf System-ffmpeg zurückfallen

PLUGIN_FFMPEG_FALLBACK_TO_SYSTEM_DESC
	EN	If the configured binary is invalid or not executable, use a system-wide ffmpeg instead.
	DE	Ist das konfigurierte Programm ungültig oder nicht ausführbar, stattdessen ein systemweites ffmpeg verwenden.

PLUGIN_FFMPEG_MODE_SECTION
	EN	Usage
	DE	Verwendung

PLUGIN_FFMPEG_MODE_SECTION_DESC
	EN	Define whether ffmpeg is only used when no native converter is available, or whether ffmpeg is always preferred.
	DE	Festlegen, ob ffmpeg nur verwendet wird, wenn kein nativer Konverter verfügbar ist, oder ob ffmpeg immer bevorzugt wird.

PLUGIN_FFMPEG_MODE_FALLBACK
	EN	fallback: native converters first, ffmpeg only if needed
	DE	Fallback: zuerst native Konverter, ffmpeg nur bei Bedarf

PLUGIN_FFMPEG_MODE_PREFER
	EN	prefer: always use ffmpeg first
	DE	Bevorzugen: immer zuerst ffmpeg verwenden

PLUGIN_FFMPEG_MODE_DESC
	EN	This changes the order of the conversion profile instances only; it does not modify any scanner or tag handling.
	DE	Dies ändert nur die Reihenfolge der Konvertierungsprofile; Scanner- oder Tag-Verarbeitung bleibt unberührt.

PLUGIN_FFMPEG_DSD_SECTION
	EN	DSD (experimental)
	DE	DSD (experimentell)

PLUGIN_FFMPEG_DSD_SECTION_DESC
	EN	Enable experimental DSD conversion (dsf/dff/WavPack DSD to FLAC at 176400Hz).
	DE	Experimentelle DSD-Konvertierung aktivieren (dsf/dff/WavPack-DSD nach FLAC mit 176400 Hz).

PLUGIN_FFMPEG_ENABLE_DSD
	EN	Enable DSD conversion
	DE	DSD-Konvertierung aktivieren

PLUGIN_FFMPEG_ENABLE_DSD_DESC
	EN	When disabled (the default), dff/dsf/wvpx ffmpeg profiles are removed and DSD playback is left to the native converters.
	DE	Wenn deaktiviert (Standard), werden dff/dsf/wvpx-ffmpeg-Profile entfernt und DSD-Wiedergabe den nativen Konvertern überlassen.
